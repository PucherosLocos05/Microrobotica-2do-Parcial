// --- Definición de Pines ---
const int ENA = 25;         // Pin PWM para el L298N
const int IN1 = 26;         // Dirección 1
const int IN2 = 27;         // Dirección 2
const int encoderPinA = 32; // Encoder (Canal A)

// --- Selector de Perfil (1 al 9) ---
// También puedes cambiarlo en vivo escribiendo un número del 1 al 9 en el Monitor Serial
int tipoPerfil = 2;

// Si es mayor a 0, se usa este valor (pulsos/s a PWM 255) y se omite la calibración.
// Después de una calibración exitosa, anota aquí el número que imprime.
const float VEL_MAX_MANUAL = 0;

volatile long encoderCount = 0;

// --- Variables de medición ---
float velMaxPPS = 1000;         // Velocidad máxima en pulsos/s
long lastPulses = 0;
unsigned long lastSpeedTime = 0;
float velocidadPPS = 0;         // Velocidad real en pulsos/s
unsigned long tInicio = 0;      // Instante en que arranca el perfil

void IRAM_ATTR countEncoder() {
  encoderCount++;
}

// Calibración: motor a PWM 255 para medir la velocidad máxima
void calibrarVelocidadMaxima() {
  Serial.println("Calibrando velocidad maxima...");
  analogWrite(ENA, 255);
  delay(1500);                         // Esperar a que el motor arranque y se estabilice

  noInterrupts();
  long inicio = encoderCount;
  interrupts();
  unsigned long t0 = millis();
  delay(1000);                         // Medir durante 1 segundo
  noInterrupts();
  long fin = encoderCount;
  interrupts();

  float medida = (fin - inicio) * 1000.0 / (millis() - t0);

  analogWrite(ENA, 0);
  delay(1000);                         // Dejar que el motor se detenga

  if (medida < 10) {
    Serial.println("ERROR: calibracion fallida (casi 0 pulsos). Revisa alimentacion del motor y el encoder.");
    Serial.println("Usando valor provisional de 1000 pulsos/s.");
    velMaxPPS = 1000;
  } else {
    velMaxPPS = medida;
    Serial.print("Velocidad maxima: ");
    Serial.print(velMaxPPS);
    Serial.println(" pulsos/s");
  }
}

// Permite cambiar de perfil escribiendo 1-9 en el Monitor Serial
void leerSerial() {
  if (Serial.available() > 0) {
    int n = Serial.parseInt();
    while (Serial.available()) Serial.read();   // Limpiar el buffer
    if (n >= 1 && n <= 9) {
      tipoPerfil = n;
      tInicio = millis();                       // Reiniciar el tiempo del perfil
      Serial.print(">> Perfil cambiado a ");
      Serial.println(tipoPerfil);
    }
  }
}

void setup() {
  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(encoderPinA, INPUT_PULLUP);

  attachInterrupt(digitalPinToInterrupt(encoderPinA), countEncoder, RISING);
  Serial.begin(115200);
  delay(500);

  // Dirección fija de giro hacia adelante
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  if (VEL_MAX_MANUAL > 0) {
    velMaxPPS = VEL_MAX_MANUAL;
    Serial.print("Usando velocidad maxima manual: ");
    Serial.print(velMaxPPS);
    Serial.println(" pulsos/s");
  } else {
    calibrarVelocidadMaxima();
  }

  // Iniciar mediciones y perfil desde cero
  noInterrupts();
  lastPulses = encoderCount;
  interrupts();
  lastSpeedTime = millis();
  tInicio = millis();
}

void loop() {
  leerSerial();

  // Tiempo en segundos desde que arrancó el perfil
  float t = (millis() - tInicio) / 1000.0;
  float pwmOutput = 0;

  switch (tipoPerfil) {

    // 1. Constante: p(t) = p0
    case 1: {
      float p0 = 120;
      pwmOutput = p0;
      break;
    }

    // 2. Rampa lineal: p(t) = p0 + at
    case 2: {
      float p0 = 0;
      float a = 20;
      pwmOutput = p0 + (a * fmod(t, 15.0));
      break;
    }

    // 3. Trapezoidal
    case 3: {
      float ciclo = fmod(t, 12.0);
      if (ciclo < 3.0) pwmOutput = ciclo * 80;
      else if (ciclo < 9.0) pwmOutput = 240;
      else pwmOutput = 240 - ((ciclo - 9.0) * 80);
      break;
    }

    // 4. Senoidal
    case 4: {
      float p0 = 125;
      float A = 100;
      float fm = 0.2;
      pwmOutput = p0 + A * sin(2 * 3.14159 * fm * t);
      break;
    }

    // 5. Escalonada (onda cuadrada)
    case 5: {
      float T = 6.0;
      if (fmod(t, T) < (T / 2.0)) pwmOutput = 60;
      else pwmOutput = 220;
      break;
    }

    // 6. Exponencial creciente
    case 6: {
      float p0 = 20;
      float alpha = 0.5;
      float k = 1.2;
      float t_mod = fmod(t, 10.0);
      pwmOutput = p0 * pow((1.0 + alpha * t_mod), k);
      break;
    }

    // 7. Gaussiano
    case 7: {
      float A = 250;
      float t0 = 5.0;
      float sigma = 1.5;
      float t_mod = fmod(t, 10.0);
      pwmOutput = A * exp(-pow((t_mod - t0) / sigma, 2));
      break;
    }

    // 8. Diente de sierra
    case 8: {
      float A = 255;
      float T = 5.0;
      pwmOutput = A * fmod(t / T, 1.0);
      break;
    }

    // 9. Ruido blanco (sin delay bloqueante: el ruido se renueva cada 100 ms)
    case 9: {
      static unsigned long lastNoise = 0;
      static float epsilon = 0;
      if (millis() - lastNoise >= 100) {
        lastNoise = millis();
        epsilon = random(-50, 50);
      }
      float p0 = 128;
      pwmOutput = p0 + epsilon;
      break;
    }
  }

  // Limitar PWM y aplicarlo
  pwmOutput = constrain(pwmOutput, 0, 255);
  analogWrite(ENA, (int)pwmOutput);

  // --- Velocidad real: pulsos por segundo, calculada cada 100 ms ---
  noInterrupts();
  long currentPulses = encoderCount;
  interrupts();

  unsigned long ahora = millis();
  if (ahora - lastSpeedTime >= 100) {
    velocidadPPS = (currentPulses - lastPulses) * 1000.0 / (ahora - lastSpeedTime);
    lastPulses = currentPulses;
    lastSpeedTime = ahora;
  }

  // --- Ambas curvas en % de la velocidad maxima ---
  float ideal = (pwmOutput / 255.0) * 100.0;
  float real  = (velocidadPPS / velMaxPPS) * 100.0;

  // Formato para Herramientas > Plotter Serie
   Serial.print("Min:0,Max:100,");   // Líneas fijas que obligan a la escala 0-100
  Serial.print("Ideal:");
  Serial.print(ideal);
  Serial.print(",Real:");
  Serial.println(real);

  delay(20);
}
