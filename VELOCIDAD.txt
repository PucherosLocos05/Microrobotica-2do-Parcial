// --- Definición de Pines ---
const int ENA = 25;       // Pin PWM para la velocidad (L298N)
const int IN1 = 26;       // Dirección 1 (L298N)
const int IN2 = 27;       // Dirección 2 (L298N)
const int potPin = 34;    // Pin analógico del potenciómetro

// --- Pines del Encoder ---
const int encoderPinA = 32; // Canal A del encoder (con interrupción)

// --- Variables de Control ---
volatile long encoderCount = 0;
unsigned long lastTime = 0;
const int interval = 100;       // Intervalo para mostrar datos en ms

// --- Modo de control ---
bool modoManual = false;        // false = potenciómetro, true = valor desde Serial
int pwmManual = 0;              // Velocidad fija recibida por Serial

void IRAM_ATTR countEncoder() {
  encoderCount++;
}

// Lee comandos desde el Monitor Serial
void leerSerial() {
  if (Serial.available() > 0) {
    String entrada = Serial.readStringUntil('\n');
    entrada.trim();
    entrada.toUpperCase();

    if (entrada.length() == 0) return;

    // Volver al potenciómetro
    if (entrada == "P" || entrada == "POT") {
      modoManual = false;
      Serial.println(">> Modo POTENCIOMETRO activado");
      return;
    }

    // Verificar que sea un número válido
    bool esNumero = true;
    for (unsigned int i = 0; i < entrada.length(); i++) {
      if (!isDigit(entrada[i])) { esNumero = false; break; }
    }

    if (esNumero) {
      int valor = entrada.toInt();
      pwmManual = constrain(valor, 0, 255);
      modoManual = true;
      Serial.print(">> Modo MANUAL: PWM fijo en ");
      Serial.println(pwmManual);
    } else {
      Serial.println(">> Comando no valido. Escribe un numero (0-255) o P para potenciometro.");
    }
  }
}

void setup() {
  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(potPin, INPUT);

  pinMode(encoderPinA, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(encoderPinA), countEncoder, RISING);

  Serial.begin(115200);

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  Serial.println("Escribe un valor (0-255) para velocidad fija, o P para usar el potenciometro.");
}

void loop() {
  // 0. Revisar si llegó algún comando por Serial
  leerSerial();

  // 1. Leer el potenciómetro
  int potValue = analogRead(potPin);

  // 2. Elegir el PWM según el modo
  int pwmValue;
  if (modoManual) {
    pwmValue = pwmManual;
  } else {
    pwmValue = map(potValue, 0, 4095, 0, 255);
  }

  // 3. Aplicar la velocidad al motor
  analogWrite(ENA, pwmValue);

  // 4. Mostrar datos cada 100 ms
  unsigned long currentTime = millis();
  if (currentTime - lastTime >= interval) {
    lastTime = currentTime;

    noInterrupts();
    long currentPulses = encoderCount;
    interrupts();

    Serial.print("Modo: ");
    Serial.print(modoManual ? "MANUAL" : "POT");
    Serial.print(" | Pot: ");
    Serial.print(potValue);
    Serial.print(" | PWM: ");
    Serial.print(pwmValue);
    Serial.print(" | Pulsos Encoder: ");
    Serial.println(currentPulses);
  }
}
